async function rejectWithdrawalRequest(requestId, reason = "") { const requestRef = doc(db, "withdrawalRequests", requestId); await runTransaction(db, async (transaction) => { const requestSnap = await transaction.get(requestRef); if (!requestSnap.exists()) { throw new Error("Request not found"); } const requestData = requestSnap.data(); if (String(requestData.status || '').toLowerCase() !== 'pending') { throw new Error("Request already processed"); } const userRef = doc(db, "users", requestData.uid); const userSnap = await transaction.get(userRef); if (!userSnap.exists()) { throw new Error("User not found"); } const currentBalance = Number(userSnap.data().balance || 0); const amount = Number(requestData.amount || 0); transaction.update(userRef, { balance: currentBalance + amount }); transaction.update(requestRef, { status: "rejected", adminNote: reason, reviewedAt: serverTimestamp(), reviewedBy: "admin" }); const userWithdrawalRef = doc(db, "users", requestData.uid, "withdrawals", requestId); transaction.update(userWithdrawalRef, { status: "rejected", adminNote: reason, reviewedAt: serverTimestamp(), reviewedBy: "admin" }); }); }
